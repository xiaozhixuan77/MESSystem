# 前端测试设计书

## 1. 概述

本文档定义工业管理 APP 前端（HarmonyOS ArkTS）的测试策略、测试用例与执行规范。

**测试框架**：@ohos/hypium（v1.0.25）
**Mock 框架**：@ohos/hamock（v1.0.0）
**关联文档**：[前端Harmony设计书](../设计书/前端Harmony设计书.md)、[前端需求分析](../需求分析/前端需求分析.md)、[前后端合并测试设计书](前后端合并测试设计书.md)

## 2. 测试策略

### 2.1 测试分层

| 层级 | 类型 | 目标 | 框架 |

## 3. 单元测试用例

### 3.1 工具函数测试（Utils.test.ets）

```typescript
import { describe, it, expect } from '@ohos/hypium';
import { PermissionUtil } from '../main/ets/utils/PermissionUtil';
import { DateUtil } from '../main/ets/utils/DateUtil';
import { FormatUtil } from '../main/ets/utils/FormatUtil';

export default function utilsTest() {
  describe('PermissionUtil', () => {
    it('admin should have all permissions', 0, () => {
      expect(PermissionUtil.hasPermission('workorder.create', 'admin')).assertTrue();
      expect(PermissionUtil.hasPermission('line.operate', 'admin')).assertTrue();
    });

    it('operator should not have dispatch permission', 0, () => {
      expect(PermissionUtil.hasPermission('workorder.dispatch', 'operator')).assertFalse();
    });

    it('leader should have view_team and dispatch', 0, () => {
      expect(PermissionUtil.hasPermission('workorder.view_team', 'leader')).assertTrue();
      expect(PermissionUtil.hasPermission('workorder.dispatch', 'leader')).assertTrue();
    });

    it('inspector should only have inspection permissions', 0, () => {
      expect(PermissionUtil.hasPermission('inspection.create', 'inspector')).assertTrue();
      expect(PermissionUtil.hasPermission('workorder.start', 'inspector')).assertFalse();
    });
  });

  describe('DateUtil', () => {
    it('should format date correctly', 0, () => {
      const date = new Date('2026-05-20T08:30:00');
      const formatted = DateUtil.format(date, 'YYYY-MM-DD HH:mm');
      expect(formatted).assertEqual('2026-05-20 08:30');
    });

    it('should calculate duration in hours', 0, () => {
      const start = new Date('2026-05-20T08:00:00');
      const end = new Date('2026-05-20T16:30:00');
      const hours = DateUtil.durationInHours(start, end);
      expect(hours).assertEqual(8.5);
    });
  });

  describe('FormatUtil', () => {
    it('should format work order code', 0, () => {
      expect(FormatUtil.formatWorkOrderCode(1)).assertEqual('MO000001');
      expect(FormatUtil.formatWorkOrderCode(100)).assertEqual('MO000100');
    });

    it('should format percentage', 0, () => {
      expect(FormatUtil.percentage(0.8567)).assertEqual('85.67%');
      expect(FormatUtil.percentage(1)).assertEqual('100.00%');
    });
  });
}
```

import { AuthApi } from '../main/ets/api/AuthApi';

export default function apiTest() {

      // Mock httpClient

        await AuthApi.login({ username: 'admin', password: 'wrong' });

      } catch (e) {
        expect((e as Error).message).assertContain('错误');
      }
    });
  });

  describe('WorkOrderApi', () => {
    it('should validate report data', 0, () => {
      // good + ng <= planQty

      const planQty = 100;
      expect(report.good + report.ng).assertLte(planQty);
    });

    it('should reject report when good + ng > planQty', 0, () => {
      const report = { good: 100, ng: 5, hours: 8 };
      const planQty = 100;
      expect(report.good + report.ng > planQty).assertTrue();
    });
  });
}

import { describe, it, expect } from '@ohos/hypium';
import { Driver, ON } from '@ohos.UiTest';
import { WebSocketService } from '@ohos/common';
import { PermissionUtil } from '@ohos/common';

export default function lineDetailTest() {
  describe('LineDetailPage', () => {
    it('should subscribe to line topics on page appear', 0, () => {
      const ws = WebSocketService.getInstance();
      let subscribed = false;

      // 页面 aboutToAppear 应订阅产线 topic
      ws.subscribe('/topic/lines/1');
      ws.subscribe('/topic/lines/1/stations');

      // 验证监听器已注册
      WebSocketService.addListener('/topic/lines/1', () => { subscribed = true; });
      expect(subscribed).assertFalse(); // 尚未触发
    });

    it('should update station status on WS message', 0, async () => {
      let updated = false;
      WebSocketService.addListener('/topic/lines/1/stations', () => {
        updated = true;
      });

      // 模拟服务端推送（通过 ws 内部 handleMessage 分发）
      const ws = WebSocketService.getInstance();
      ws.handleTestMessage({
        topic: '/topic/lines/1/stations',
        version: 1715762340123,
        eventId: 'evt-test-001',
        payload: {
          lineId: 1,
          stations: [{ id: 1, deviceStatus: '运行中', rawCount: 0, finishedCount: 0, ngCount: 0 }]
        }
      });

      expect(updated).assertTrue();
    });

    it('should disable buttons for inspector role', 0, () => {
      // 质检员无产线操作权限

      await reportBtn.click();
      await driver.delayMs(500);

## 6. WebSocket 测试

```typescript
import { describe, it, expect } from '@ohos/hypium';
import { WebSocketService } from '../main/ets/api/WebSocketService';

export default function websocketTest() {
  describe('WebSocketService', () => {
    it('should connect successfully', 0, async () => {
      const ws = WebSocketService.getInstance();
      ws.connect('ws://localhost:8080/ws');
      // 等待连接完成（isConnected 由 @Trace 观测）
      await new Promise(resolve => setTimeout(resolve, 1000));
      expect(ws.isConnected).assertTrue();
    });

    it('should subscribe and receive messages', 0, async () => {
      const ws = WebSocketService.getInstance();
      let received = false;
      
      // 使用静态监听器注册（与设计书 V2 架构一致）
      WebSocketService.addListener('/topic/lines/1', () => {
        received = true;
      });
      
      ws.subscribe('/topic/lines/1');
      // 模拟服务端推送
      ws.handleTestMessage({
        topic: '/topic/lines/1',
        version: 1234567890,
        eventId: 'evt-test-001',
        payload: { status: '生产中' }
      });
      expect(received).assertTrue();
    });

    it('should unsubscribe on page leave', 0, () => {
      const ws = WebSocketService.getInstance();
      ws.subscribe('/topic/lines/1');
      ws.unsubscribe('/topic/lines/1');
      WebSocketService.removeListener('/topic/lines/1');
      // 验证监听器已从 Map 中移除
    });

    it('should reconnect after disconnect', 0, async () => {
      const ws = WebSocketService.getInstance();
      // 手动触发 close 回调模拟断线
      ws.handleTestClose();
      await new Promise(resolve => setTimeout(resolve, 6000));
      // 验证重连后 isConnected 恢复
      expect(ws.isConnected).assertTrue();
    });
  });
}
```

## 7. 性能测试

| 测试项 | 指标 | 测试方法 |
|--------|------|---------|
| 页面启动时间 | < 1s | 记录 aboutToAppear 到首次渲染完成时间 |

# 后端测试设计书

## 1. 概述

本文档定义工业管理 APP 后端（Spring Boot）的测试策略、测试用例与执行规范。

**测试框架**：JUnit 5 + Mockito + AssertJ
**数据库测试**：H2（内存数据库）/ Testcontainers（MySQL）
**WebSocket 测试**：Spring WebSocket Test
**关联文档**：[后端SpringBoot设计书](../设计书/后端SpringBoot设计书.md)、[后端需求分析](../需求分析/后端需求分析.md)、[前后端合并测试设计书](前后端合并测试设计书.md)

class LineSimulatorServiceTest {

    @Mock
    private LineBroadcaster broadcaster;

        ReflectionTestUtils.setField(simulatorService, "logMapper", logMapper);
        ReflectionTestUtils.setField(simulatorService, "broadcaster", broadcaster);
        ReflectionTestUtils.setField(simulatorService, "runningSimulators", runningSimulators);
        ReflectionTestUtils.setField(simulatorService, "scheduler",
            Executors.newSingleThreadScheduledExecutor());
        ReflectionTestUtils.setField(simulatorService, "goodRate", 0.95);
    }

    @Test
    @DisplayName("启动仿真：成功挂起定时任务")
    void startSimulation_shouldScheduleTask() {
        simulatorService.startSimulation(1L, 100L);

        assertThat(runningSimulators).containsKey(1L);
    }

    @Test
    @DisplayName("重复启动同一产线应失败")
    void startSimulation_shouldFail_whenAlreadyRunning() {
        runningSimulators.put(1L, mock(ScheduledFuture.class));

        assertThatThrownBy(() -> simulatorService.startSimulation(1L, 100L))
            .isInstanceOf(BusinessException.class)
            .hasMessageContaining("已在仿真中");
    }

    @Test
    @DisplayName("停止仿真：取消定时任务")
    void stopSimulation_shouldCancelTask() {
        ScheduledFuture<?> future = mock(ScheduledFuture.class);
        runningSimulators.put(1L, future);

        simulatorService.stopSimulation(1L);

        verify(future).cancel(false);
        assertThat(runningSimulators).doesNotContainKey(1L);
    }

    @Test
    @DisplayName("Tick 应更新工作站计数并广播（通过反射调用私有方法）")
    void tick_shouldUpdateCountersAndBroadcast() throws Exception {
        LineSimulator simulator = new LineSimulator(1L, 100L, 0.95);
        List<StationState> states = Arrays.asList(
            new StationState("WS-FLX-OP10", 10, 9, 1)
        );
        // 使用 spy 模拟 simulator
        LineSimulator spySimulator = spy(simulator);
        when(spySimulator.tick()).thenReturn(states);

        // 反射调用私有 tick 方法
        Method tickMethod = LineSimulatorService.class.getDeclaredMethod("tick", Long.class, LineSimulator.class);
        tickMethod.setAccessible(true);
        tickMethod.invoke(simulatorService, 1L, spySimulator);

        verify(logMapper).insert(any(ProductionLog.class));
        verify(broadcaster).broadcastStationUpdate(1L, states);
    }
}

@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
class WebSocketIntegrationTest {

    @LocalServerPort
    private int port;

    @Autowired
    private LineBroadcaster broadcaster;

    private WebSocketStompClient stompClient;
    private StompSession session;

    @BeforeEach
    void setUp() {
        stompClient = new WebSocketStompClient(
            new StandardWebSocketClient()
        );
    }

    @Test
    @DisplayName("订阅产线状态并接收广播")
    void subscribeLineStatus_shouldReceiveBroadcast() throws Exception {
        BlockingQueue<Object> messages = new LinkedBlockingQueue<>();

        StompSessionHandler handler = new StompSessionHandlerAdapter() {
            @Override
            public void afterConnected(StompSession session, StompHeaders headers) {
                session.subscribe("/topic/lines/1", new StompFrameHandler() {
                    @Override
                    public Type getPayloadType(StompHeaders headers) {
                        return LineStatusVO.class;
                    }
                    @Override
                    public void handleFrame(StompHeaders headers, Object payload) {
                        messages.add(payload);
                    }
                });
            }
        };

        session = stompClient.connect(
            "ws://localhost:" + port + "/ws", handler).get(5, TimeUnit.SECONDS);

        // 广播消息
        LineStatusVO status = new LineStatusVO();
        status.setLineId(1L);
        status.setStatus("生产中");
        broadcaster.broadcastLineStatus(1L, status);

        // 验证接收
        LineStatusVO received = (LineStatusVO) messages.poll(5, TimeUnit.SECONDS);
        assertThat(received).isNotNull();
        assertThat(received.getStatus()).isEqualTo("生产中");
    }

    @Test
    @DisplayName("未授权用户连接应被拒绝")
    void connectWithoutAuth_shouldBeRejected() throws Exception {
        StompSessionHandler handler = new StompSessionHandlerAdapter() {
            @Override
            public void handleTransportError(StompSession session, Throwable exception) {
                assertThat(exception).isNotNull();
            }
        };

        // 不带 Token 连接应失败
        assertThatThrownBy(() -> stompClient.connect(
            "ws://localhost:" + port + "/ws", handler).get(5, TimeUnit.SECONDS))
            .isInstanceOfAny(ExecutionException.class, MessageDeliveryException.class);
    }

    @Test
    @DisplayName("过期 Token 连接应被拒绝")
    void connectWithExpiredToken_shouldBeRejected() throws Exception {

            @Override

            }
        };

        StompHeaders expiredHeaders = new StompHeaders();

            "ws://localhost:" + port + "/ws", handler, expiredHeaders)

@SpringBootTest
class StationRedisCacheTest {

    @Container

    private WorkstationMapper workstationMapper;

    @DisplayName("工作站状态写入 Redis 后可读取")
    void writeAndReadStationStatus() {

        status.setStationId(1L);
        status.setStatus("running");
        status.setRawCount(100);
        status.setFinishedCount(95);

        stationCacheService.updateStatus(1L, status);

        assertThat(cached).isNotNull();
        assertThat(cached.getStatus()).isEqualTo("running");
        assertThat(cached.getFinishedCount()).isEqualTo(95);
    }

    @Test
    @DisplayName("Redis 不可用时降级读取 MySQL")
    void fallbackToMySQL_whenRedisDown() {
        // 停止 Redis 容器
        redis.stop();

        Workstation ws = new Workstation();
        ws.setId(1L);
        ws.setStatus("paused");
        ws.setRawCount(50);
        workstationMapper.insert(ws);

        StationStatusVO result = stationCacheService.getStatus(1L);

        assertThat(result).isNotNull();
        assertThat(result.getStatus()).isEqualTo("paused");
    }

    @Test
    @DisplayName("启动时预加载所有工作站状态到 Redis")
    void preloadOnStartup() {
        stationCacheService.preloadAllStations();

        List<Workstation> allStations = workstationMapper.selectList(null);
        for (Workstation ws : allStations) {
            StationStatusVO cached = stationCacheService.getStatus(ws.getId());
            assertThat(cached).isNotNull();
            assertThat(cached.getStatus()).isEqualTo(ws.getStatus());
        }
    }

    @Test
    @DisplayName("缓存与 MySQL 5s 批量 flush 一致性")
    void cacheAndMySQLConsistency() throws InterruptedException {
        StationStatusVO status = new StationStatusVO();
        status.setStationId(2L);
        status.setRawCount(10);

        stationCacheService.updateStatus(2L, status);
        // Redis 立即有值
        assertThat(stationCacheService.getStatus(2L).getRawCount()).isEqualTo(10);

        // MySQL 可能在 5s 批量写入后才更新

@SpringBootTest
class PerformanceBenchmarkTest {

    @Autowired
    private WorkOrderService workOrderService;

    @Test
    @DisplayName("工单列表查询性能：P95 < 200ms")
    void workOrderListQueryPerformance() {

        }

        assertThat(avg).isLessThan(200);
    }

    @Test
    @DisplayName("并发状态变更：只有一个成功")
    void concurrentStateChange_shouldOnlySucceedOnce() throws InterruptedException {
        int threadCount = 10;
        CountDownLatch latch = new CountDownLatch(threadCount);
        AtomicInteger successCount = new AtomicInteger(0);

        for (int i = 0; i < threadCount; i++) {
            new Thread(() -> {
                try {
                    workOrderService.start(1L);
                    successCount.incrementAndGet();
                } catch (BusinessException e) {
                    // 预期大部分失败
                } finally {
                    latch.countDown();
                }
            }).start();
        }

        latch.await(10, TimeUnit.SECONDS);
        assertThat(successCount.get()).isEqualTo(1);
    }
}

## 6. 安全测试

| 测试项 | 目标 | 方法 |
|--------|------|------|
